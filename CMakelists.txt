cmake_minimum_required(VERSION 3.15)

set(CMAKE_C_COMPILER "D:/msys64/ucrt64/bin/gcc.exe" )
set(CMAKE_CXX_COMPILER "D:/msys64/ucrt64/bin/g++.exe" )
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static")

project(Showcase VERSION 1.0
                  DESCRIPTION "Showcase Project"
                  LANGUAGES CXX)

#set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

make_directory( ${CMAKE_SOURCE_DIR}/build )

find_package(Vulkan REQUIRED)
include_directories( ${VULKAN_INCLUDE_DIRS} )

#find_package(glfw3 REQUIRED)

include_directories(include)

add_executable(showcase src/main.cpp)

include(FetchContent)

set(FETCHCONTENT_BASE_DIR ${PROJECT_SOURCE_DIR}/libs CACHE PATH "Missing description." FORCE)

FetchContent_Declare(
	glm
	GIT_REPOSITORY	https://github.com/g-truc/glm.git
	GIT_TAG 	0af55ccecd98d4e5a8d1fad7de25ba429d60e863 #refs/tags/1.0.1
)

FetchContent_Declare(assimp
GIT_REPOSITORY https://github.com/assimp/assimp.git
GIT_TAG master)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INJECT_DEBUG_POSTFIX OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(glm)
FetchContent_MakeAvailable(assimp)

target_include_directories(showcase PRIVATE libs/assimp-src/include)

target_include_directories(showcase PRIVATE include/GLFW)
target_link_directories(showcase PRIVATE lib/GLFW)

target_link_libraries(showcase PRIVATE glfw3 Vulkan::Vulkan assimp glm -static-libgcc -static-libstdc++)
